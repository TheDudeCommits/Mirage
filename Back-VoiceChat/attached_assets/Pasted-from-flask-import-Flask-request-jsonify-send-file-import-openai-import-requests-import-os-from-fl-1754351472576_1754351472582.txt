from flask import Flask, request, jsonify, send_file
import openai
import requests
import os
from flask_cors import CORS

app = Flask(__name__)
CORS(app)

openai.api_key = os.environ.get("OPENAI_API_KEY")
ELEVENLABS_API_KEY=<set-in-local-env>
VOICE_ID=<set-in-local-env>

@app.route('/api/voice', methods=['POST'])
def handle_voice():
    if 'audio' not in request.files:
        return jsonify({'error': 'No audio file uploaded'}), 400

    audio_file = request.files['audio']
    temp_path = "temp_input.mp3"
    audio_file.save(temp_path)

    try:
        # Step 1: Transcribe using Whisper
        transcript = openai.Audio.transcribe("whisper-1", open(temp_path, "rb"))
        user_text = transcript['text']
        print("Transcription:", user_text)

        # Step 2: Get response from GPT
        response = openai.ChatCompletion.create(
            model="gpt-4",
            messages=[{"role": "user", "content": user_text}]
        )
        ai_reply = response['choices'][0]['message']['content']
        print("GPT Reply:", ai_reply)

        # Step 3: Convert GPT reply to speech with ElevenLabs
        tts_url = f"https://api.elevenlabs.io/v1/text-to-speech/{VOICE_ID}"
        headers = {
            "xi-api-key": ELEVENLABS_API_KEY,
            "Content-Type": "application/json"
        }
        payload = {
            "text": ai_reply,
            "model_id": "eleven_monolingual_v1",
            "voice_settings": {
                "stability": 0.5,
                "similarity_boost": 0.8
            }
        }

        tts_response = requests.post(tts_url, headers=headers, json=payload)
        audio_output_path = "mira_reply.mp3"

        with open(audio_output_path, "wb") as f:
            f.write(tts_response.content)

        return send_file(audio_output_path, mimetype="audio/mpeg", as_attachment=False), 200

    except Exception as e:
        return jsonify({"error": str(e)}), 500

if __name__ == '__main__':
    app.run(host="0.0.0.0", port=3000)
